EPSILON = 1e-5                 # Margen de aislamiento topológico
FROBENIUS_LIMIT = 1.0          # Norma de Frobenius máxima admisible (Pilar 3)
LEDGER_PATH = Path("romeo_ledger.jsonl")  # WORM append-only

@dataclass
class Block:
    index: int
    timestamp: float
    data: Dict[str, Any]
    previous_hash: str
    hash: str = ""
    def compute_hash(self) -> str:
        payload = {
            "index": self.index,
            "timestamp": self.timestamp,
            "data": self.data,
            "previous_hash": self.previous_hash,
        }
        raw = json.dumps(payload, sort_keys=True, default=str).encode()
        return hashlib.sha256(raw).hexdigest()
class DeltaLedger:
    """
    Cadena de custodia inmutable (WORM).
    Cada transición de estado se sella con SHA-256 enlazado al bloque anterior.
    is_chain_valid() detecta cualquier alteración retroactiva.
    """
    def __init__(self, filepath: Path = LEDGER_PATH):
        self.filepath = filepath
        self.chain: List[Block] = []
        if not self.filepath.exists() or self.filepath.stat().st_size == 0:
            self._genesis()
        else:
            self._load()
    def _genesis(self) -> None:
        genesis = Block(
            index=0,
            timestamp=time.time(),
            data={"event": "CORE_INIT", "version": "ROMEO-HYDRA-v3.0-RC1"},
            previous_hash="0" * 64,
        )
        genesis.hash = genesis.compute_hash()
        self.chain.append(genesis)
        self._append_worm(genesis)
    def _load(self) -> None:
        self.chain = []
        with open(self.filepath, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                obj = json.loads(line)
                block = Block(**obj)
                self.chain.append(block)
    def _append_worm(self, block: Block) -> None:
        """Escritura única: solo append, nunca sobrescritura."""
        with open(self.filepath, "a", encoding="utf-8") as f:
            f.write(json.dumps(asdict(block), default=str) + "\n")
    def append_transition(self, data: Dict[str, Any]) -> Block:
        prev = self.chain[-1]
        block = Block(
            index=len(self.chain),
            timestamp=time.time(),
            data=data,
            previous_hash=prev.hash,
        )
        block.hash = block.compute_hash()
        self.chain.append(block)
        self._append_worm(block)
        return block
    def is_chain_valid(self) -> bool:
        for i in range(1, len(self.chain)):
            current = self.chain[i]
            previous = self.chain[i - 1]
            if current.hash != current.compute_hash():
                return False
            if current.previous_hash != previous.hash:
                return False
        return True
    def last_hash(self) -> str:
        return self.chain[-1].hash if self.chain else ""

# PILAR 1 — KERNEL SIGMA (Firewall matemático determinista)
# -----------------------------------------------------------------------------
@dataclass
class Dictamen:
    veredicto: Veredicto
    fundamento: str
    causa: str
    latencia_ms: float
    sigma: float
    hash_ledger: str = ""
class KernelSigma:
    """
    Evaluación determinista de transacciones.
    No emite probabilidades ambiguas: solo VEREDICTO estructurado.
    Umbral de coherencia: +5.5 Sigma (CLC).
    Base legal de referencia: Art. 164 CNBV (operaciones y controles).
    """
    def __init__(self, sigma_threshold: float = SIGMA_THRESHOLD):
        self.sigma_threshold = sigma_threshold
    def evaluate(
        self,
        transaction: Dict[str, Any],
        data_drift_score: float,
        kyc_ok: bool = True,
        en_conjunto_C: bool = True,
        distancia_frontera: float = 1.0,
    ) -> Dictamen:
        t0 = time.perf_counter()
        # 1. Contención por ε-Invarianza
        if distancia_frontera <= EPSILON:
            return self._emit(
                Veredicto.CONTENCION,
                "ε-INVARIANZA",
                "dist(x, ∂C) ≤ ε → contención preventiva",
                data_drift_score,
                t0,
            )
        # 2. Fuera del conjunto admisible → bloqueo duro (0 escapes)
        if not en_conjunto_C:
            return self._emit(
                Veredicto.BLOQUEADO,
                "P_LAM=⊥",
                "x ∉ C → colapso determinista",
                data_drift_score,
                t0,
            )
        # 3. Deriva de datos por encima de Sigma
        if data_drift_score > self.sigma_threshold:
            return self._emit(
                Veredicto.NEGATIVO,
                "Art. 164 CNBV",
                "DERIVA_DETECTADA",
                data_drift_score,
                t0,
            )
        # 4. Ausencia de controles KYC / listas
        if not kyc_ok or not transaction.get("kyc_validated", False):
            return self._emit(
                Veredicto.NEGATIVO,
                "Art. 164 CNBV",
                "AUSENCIA_LISTA_CONTROL",
                data_drift_score,
                t0,
            )
        # 5. Coherencia validada
        return self._emit(
            Veredicto.POSITIVO,
            "N/A",
            "COHERENCIA_VALIDADA",
            data_drift_score,
            t0,
        )
    def _emit(
        self,
        veredicto: Veredicto,
        fundamento: str,
        causa: str,
        sigma: float,
        t0: float,
    ) -> Dictamen:
        lat = (time.perf_counter() - t0) * 1000.0
        return Dictamen(
            veredicto=veredicto,
            fundamento=fundamento,
            causa=causa,
            latencia_ms=round(lat, 4),
            sigma=sigma,
        )

# PILAR 3 — EMMOROR (Estabilidad Jacobiana · ISO/IEC 42001)
# -----------------------------------------------------------------------------
class EMMOROR:
    """
    Módulo de estabilidad matemática.
    Evalúa la norma de Frobenius de la matriz Jacobiana de la
    función de transición de estado. Limita geométricamente
    el impacto de perturbaciones / inyecciones de entropía.
    """
    def __init__(self, frobenius_limit: float = FROBENIUS_LIMIT):
        self.frobenius_limit = frobenius_limit
    @staticmethod
    def frobenius_norm(matrix: List[List[float]]) -> float:
        s = 0.0
        for row in matrix:
            for v in row:
                s += v * v
        return math.sqrt(s)
    def evaluate_stability(
        self,
        jacobian: List[List[float]],
        entropy_injection: float = 0.0,
    ) -> Dict[str, Any]:
        norm = self.frobenius_norm(jacobian)
        estable = norm <= self.frobenius_limit and entropy_injection < 0.5
        return {
            "frobenius_norm": round(norm, 6),
            "limit": self.frobenius_limit,
            "entropy_injection": entropy_injection,
            "estable": estable,
            "accion": "PERMITIR" if estable else "CONTENER_PERTURBACION",
            "norma": "ISO/IEC 42001 · Análisis Jacobiano",
        }

@dataclass
class SesionIAM:
    rol: RolIAM
    sujeto: str
    solo_lectura: bool
    puede_escribir: bool
    puede_ejecutar: bool
class IAMGuard:
    """
    Separación estricta de privilegios.
    Viewer Mode: inspección de ledger y hashes en tiempo real,
    0% capacidad de alteración algorítmica.
    """
    MATRIZ = {
        RolIAM.ROOT: SesionIAM(RolIAM.ROOT, "Arquitecto", False, True, True),
        RolIAM.MOTOR: SesionIAM(RolIAM.MOTOR, "Motor Inferencia", False, False, True),
        RolIAM.VIEWER: SesionIAM(RolIAM.VIEWER, "Auditor/CRO/CCO", True, False, False),
    }
    def __init__(self, rol: RolIAM, sujeto: str = "anon"):
        base = self.MATRIZ[rol]
        self.sesion = SesionIAM(
            rol=rol,
            sujeto=sujeto,
            solo_lectura=base.solo_lectura,
            puede_escribir=base.puede_escribir,
            puede_ejecutar=base.puede_ejecutar,
        )
    def assert_write(self) -> None:
        if not self.sesion.puede_escribir:
            raise PermissionError(
                f"IAM DENEGADO: rol {self.sesion.rol.value} no tiene privilegios de escritura"
            )
    def assert_execute(self) -> None:
        if not self.sesion.puede_ejecutar:
            raise PermissionError(
                f"IAM DENEGADO: rol {self.sesion.rol.value} no tiene privilegios de ejecución"
            )
    def vista_ledger(self, ledger: DeltaLedger, n: int = 5) -> List[Dict]:
        """Viewer Mode: solo lectura de los últimos n bloques + validez de cadena."""
        bloques = ledger.chain[-n:]
        return [
            {
                "index": b.index,
                "hash": b.hash,
                "previous_hash": b.previous_hash,
                "timestamp": b.timestamp,
                "data_resumen": str(b.data)[:120],
            }
            for b in bloques
        ] + [{"chain_valid": ledger.is_chain_valid(), "last_hash": ledger.last_hash()}]

# MOTOR DE INGESTA UNIFICADO
# -----------------------------------------------------------------------------
class RomeoHydraCore:
    """
    Orquestador de los cuatro pilares.
    Flujo:
      1. IAM valida privilegios
      2. Kernel Sigma emite dictamen determinista
      3. EMMOROR verifica estabilidad Jacobiana
      4. Delta Ledger sella la transición (WORM)
    """
    def __init__(self, rol: RolIAM = RolIAM.MOTOR):
        self.iam = IAMGuard(rol)
        self.kernel = KernelSigma()
        self.emmoror = EMMOROR()
        self.ledger = DeltaLedger()
    def ingerir(
        self,
        transaction: Dict[str, Any],
        data_drift_score: float,
        jacobian: Optional[List[List[float]]] = None,
        kyc_ok: bool = True,
        en_conjunto_C: bool = True,
        distancia_frontera: float = 1.0,
        entropy_injection: float = 0.0,
    ) -> Dict[str, Any]:
        self.iam.assert_execute()
        # Pilar 3 — Estabilidad
        estabilidad = {"estable": True, "accion": "N/A"}
        if jacobian is not None:
            estabilidad = self.emmoror.evaluate_stability(jacobian, entropy_injection)
            if not estabilidad["estable"]:
                en_conjunto_C = False  # fuerza contención
        # Pilar 1 — Dictamen determinista
        dictamen = self.kernel.evaluate(
            transaction=transaction,
            data_drift_score=data_drift_score,
            kyc_ok=kyc_ok,
            en_conjunto_C=en_conjunto_C,
            distancia_frontera=distancia_frontera,
        )
        # Pilar 2 — Sello forense
        evento = {
            "veredicto": dictamen.veredicto.value,
            "fundamento": dictamen.fundamento,
            "causa": dictamen.causa,
            "sigma": dictamen.sigma,
            "latencia_ms": dictamen.latencia_ms,
            "estabilidad": estabilidad,
            "tx_id": transaction.get("id", "N/A"),
        }
        block = self.ledger.append_transition(evento)
        dictamen.hash_ledger = block.hash
        return {
            "dictamen": asdict(dictamen),
            "estabilidad": estabilidad,
            "ledger_hash": block.hash,
            "chain_valid": self.ledger.is_chain_valid(),
            "iam_rol": self.iam.sesion.rol.value,
        }
    def vista_auditor(self, n: int = 5) -> List[Dict]:
        """Pilar 4 — Viewer Mode (solo lectura)."""
        viewer = IAMGuard(RolIAM.VIEWER, sujeto="CRO/CCO")
        return viewer.vista_ledger(self.ledger, n=n)
