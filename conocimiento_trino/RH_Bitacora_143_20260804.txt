```python
import math
import logging
# Configuración del log de Gobernanza
logging.basicConfig(level=logging.INFO, format='%(asctime)s - GAMMA - %(levelname)s - %(message)s')
class GammaRegulator:
    """
    Régimen Gamma: Motor de Gobernanza de Entropía.
    Responsable de calcular el Índice de Estabilidad de Hipótesis (HSI)
    y decidir sobre la persistencia de estados.
    """
    def __init__(self, entropy_threshold=0.75):
        self.entropy_threshold = entropy_threshold
        logging.info("Régimen Gamma inicializado. Umbral de estabilidad: %s", self.entropy_threshold)
    def calculate_entropy(self, data_points):
        """
        Calcula la entropía de Shannon básica para evaluar la dispersión de datos.
        """
        if not data_points:
            return 0
        # Normalización simple para el cálculo
        total = sum(data_points)
        probs = [p / total for p in data_points if p > 0]
        entropy = -sum(p * math.log2(p) for p in probs)
        return entropy
    def evaluate_hsi(self, friction_index, entropy):
        """
        Calcula el Índice de Estabilidad de Hipótesis (HSI).
        Combina la fricción (Beta) y la entropía (Gamma).
        """
        # HSI normalizado (0 a 1). 1 es máxima estabilidad.
        hsi = 1 - (friction_index * 0.5 + (entropy / 10) * 0.5)
        return max(0, min(1, hsi))
    def decide(self, friction_index, data_points):
        """
        Toma de decisiones autónoma (Gobernanza).
        """
        entropy = self.calculate_entropy(data_points)
        hsi = self.evaluate_hsi(friction_index, entropy)
        logging.info(f"Evaluación: Fricción={friction_index}, Entropía={entropy:.2f}, HSI={hsi:.2f}")
        if hsi < self.entropy_threshold:
            return "PRUNE", hsi
        else:
            return "CONTINUE", hsi
# Ejemplo de uso interno (para pruebas de validación)
if __name__ == "__main__":
    regulator = GammaRegulator()
    # Simulación: Fricción baja (0.2), Entropía baja (0.5)
    decision, hsi = regulator.decide(0.2, [10, 12, 11])
    print(f"Decisión: {decision} con HSI: {hsi}")
```
