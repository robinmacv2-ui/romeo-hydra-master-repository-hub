class JacobianoCoherencia {
    [double]$Threshold
    [double]$PenaltyFactor
    JacobianoCoherencia([double]$threshold = 0.85) {
        $this.Threshold = $threshold
        $this.PenaltyFactor = 2.5
    }
    # Calcula la Norma de Frobenius: sqrt(sum(abs(J_ij)^2))
    [double] CalcularVolatilidad([double[,]]$MatrizJ) {
        $sumaCuadrados = 0
        $filas = $MatrizJ.GetLength(0)
        $columnas = $MatrizJ.GetLength(1)
        for ($i = 0; $i -lt $filas; $i++) {
            for ($j = 0; $j -lt $columnas; $j++) {
                $sumaCuadrados += [Math]::Pow($MatrizJ[$i, $j], 2)
            }
        }
        return [Math]::Sqrt($sumaCuadrados)
    }
    [PSCustomObject] EvaluarEstabilidad([double[,]]$MatrizJ) {
        $volatilidad = $this.CalcularVolatilidad($MatrizJ)
        $alerta = $false
        $factor = 1.0
        if ($volatilidad -gt $this.Threshold) {
            Write-Host "[REGIMEN BETA ALERT] Volatilidad detectada: $volatilidad" -ForegroundColor Red
            $alerta = $true
            $factor = $volatilidad * $this.PenaltyFactor
        }
        return [PSCustomObject]@{
            AlertaDetected = $alerta
            PenaltyScale   = $factor
            MetricValue    = $volatilidad
        }
    }
}
