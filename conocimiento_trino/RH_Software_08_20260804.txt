def romeo_hpr(observations, alpha, sigma_init, tol=1e-6, max_iter=1000):
    """
    Operador central de recuperación de potencial (ROMEO-HPR).
    """
    sigma = sigma_init
    for k in range(max_iter):
        # 1. Forward pass: Generar trayectorias desde el potencial actual
        x_sim = integrate_dynamics(sigma)
        
        # 2. Evaluación: L = desacuerdo, S = regularización Hessiana
        loss = compute_l2_loss(observations, x_sim)
        reg = compute_hessian_norm(sigma)
        j_alpha = loss + alpha * reg
        
        # 3. Optimización: Descenso variacional
        grad = compute_gradient(j_alpha)
        if norm(grad) < tol:
            break
            
        # 4. Proyección: Asegurar convexidad (Sigma \in P_A)
        sigma = project_to_convex_cone(sigma - learning_rate * grad)
