import numpy as np

class BetaRegime:
    """
    Régimen Beta: Implementación del Jacobiano de Amortiguación.
    Monitorea la volatilidad diferencial entre estados de inferencia.
    """
    def __init__(self, sensitivity_threshold=0.5):
        self.last_state = None
        self.sensitivity_threshold = sensitivity_threshold
        self.dampening_factor = 1.0  # Factor de escala (1.0 = sin fricción)

    def _calculate_volatility(self, current_vector):
        """Calcula la distancia Euclidiana entre estados consecutivos."""
        if self.last_state is None:
            return 0.0
        
        # Vectores normalizados para cálculo preciso
        current = np.array(current_vector)
        last = np.array(self.last_state)
        return float(np.linalg.norm(current - last))

    def process_state(self, inference_vector, coherence_score):
        """
        Calcula el factor de amortiguación dinámico.
        Si la volatilidad es alta y la coherencia (Alpha) baja, aplicamos dampening.
        """
        volatility = self._calculate_volatility(inference_vector)
        self.last_state = inference_vector
        
        # Jacobiano de Amortiguación:
        # Penalización si volatilidad > umbral Y la coherencia de Alpha es insuficiente.
        if volatility > self.sensitivity_threshold and coherence_score < 0.7:
            self.dampening_factor = max(0.1, 1.0 - (volatility * 0.5))
        else:

            # Recuperación gradual de la estabilidad
            self.dampening_factor = min(1.0, self.dampening_factor + 0.1)
            
        return {
            "volatility": volatility,
            "dampening_factor": round(self.dampening_factor, 4),
            "status": "stable" if self.dampening_factor > 0.8 else "dampened"
        }
