### Paso 2: El Módulo de Código (src/lib/sigmaOptimizer.ts)
Crea este archivo en la estructura de tu proyecto (src/lib/sigmaOptimizer.ts). Esta función aplica una proyección convexa básica (clamping y normalización de pesos) para garantizar que cualquier métrica de riesgo o desvío se mantenga dentro de un espacio seguro antes de ser guardada en Firebase:
```typescript
// ROMEO-HYDRA: Regimen Sigma - Optimizador de Espacio Convexo
export interface InferenceState {
  riskScore: number; // Debe estar entre 0 y 1
  popperIndex: number; // Índice de falsación (idealmente > 0.85)
  driftValue: number;
}
export interface OptimizationResult {
  optimizedState: InferenceState;
  isFeasible: boolean;
  actionFlag: string;
}
/**
 * Proyecta el estado actual en el conjunto convexo de restricciones operativas.
 */
otyep: // (fixing typo in thought process -> export function)
export function optimizeConvexState(state: InferenceState): OptimizationResult {
  // Definición del Espacio Convexa (Límites estrictos)
  const MAX_RISK = 0.05;
  const MIN_POPPER = 0.85;
  let feasible = true;
  let flag = "[OPTIMIZADO_POR_SIGMA: NORMAL]";
  // Restricción 1: Proyección convexa del riesgo (Clamping al hiperplano seguro)
  let clampedRisk = state.riskScore;
  if (clampedRisk > MAX_RISK) {
    clampedRisk = MAX_RISK;
    feasible = false;
    flag = "[OPTIMIZADO_POR_SIGMA: RISK_CLIPPED]";
  }
  // Restricción 2: Verificación del índice Popperiano
  let adjustedPopper = state.popperIndex;
  if (adjustedPopper < MIN_POPPER) {
    adjustedPopper = MIN_POPPER;
    feasible = false;
    flag = "[OPTIMIZADO_POR_SIGMA: POPPER_ENFORCED]";
  }
  return {
    optimizedState: {
      riskScore: clampedRisk,
      popperIndex: adjustedPopper,
      driftValue: state.driftValue
    },
    isFeasible: feasible,
    actionFlag: flag
  };
}
```
