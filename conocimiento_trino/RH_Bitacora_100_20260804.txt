def validar_transaccion(data):
    if not data or not isinstance(data, dict):
        return False, "Estructura de datos inválida"
    # Validar campos obligatorios y firmas/integridad simuladas
    campos_requeridos = ["id", "payload", "checksum"]
    for campo in campos_requeridos:
        if campo not in data:
            return False, f"Falta el campo obligatorio: {campo}"
    # Simulación de validación criptográfica / corrupción
    payload = str(data.get("payload", ""))
    checksum_recibido = data.get("checksum", "")
    # Calcular checksum esperado para verificar integridad
    checksum_calculado = hashlib.sha256(payload.encode()).hexdigest()[:16]
    if checksum_recibido != checksum_calculado:
        return False, "Falla de integridad: Checksum inválido o transacción corrupta"
    # Validaciones adicionales de negocio o anomalías
    if "corrupt" in payload.lower() or data.get("anomalia", False):
        return False, "Transacción marcada como anómala o corrupta"
    return True, "Transacción legítima y validada"

def evaluar_coherencia(matriz):
    """
    Función interna (proxy JacobianoCoherencia) que calcula 
    la estabilidad basándose en la norma de Frobenius.
    """
    # Si la matriz es singular o tiene determinante 0, aplicamos penalización
    det = np.linalg.det(matriz)
    norma_fro = np.linalg.norm(matriz)
    if abs(det) < 1e-9:
        is_stable = False
        # Simulación del cálculo dinámico del Kernel Sigma frente al colapso espacial
        penalty_factor = float(norma_fro * 2.5)
    else:
        is_stable = True
        penalty_factor = 1.0
    return is_stable, float(norma_fro), penalty_factor
