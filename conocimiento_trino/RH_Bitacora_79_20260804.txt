import numpy as np

from core.beta import JacobianoCoherencia
app = Flask(__name__)
# Inicializamos el motor matemático
coherencia_motor = JacobianoCoherencia()
@app.route('/audit', methods=['POST'])
def audit_transaction():
    try:
        data = request.get_json()
        # Extraemos la matriz enviada en la transacción
        # Si no envían una, usamos una matriz vacía o por defecto
        matrix_data = data.get('matrix', [[0.1, 0.2], [0.3, 0.1]])
        matriz = np.array(matrix_data)
        # Evaluamos la estabilidad usando el Jacobiano
        inestable, factor_penalizacion = coherencia_motor.evaluar_estabilidad(matriz)
        if inestable:
            return jsonify({
                "status": "rejected",
                "message": "Alerta de seguridad: Inestabilidad detectada en la métrica convexa.",
                "factor_penalizacion": float(factor_penalizacion),
                "anomalies_detected": True
            }), 400
        else:
            return jsonify({
                "status": "success",
                "message": "Transacción validada y aprobada por el motor matemático.",
                "factor_penalizacion": float(factor_penalizacion),
                "anomalies_detected": False
            }), 200

        return jsonify({
            "status": "error",
            "message": "Error al procesar la matriz en el servidor.",
            "error_detail": str(e),
            "anomalies_detected": True
        }), 500
if __name__ == '__main__':

from flask import Flask, request, jsonify
import numpy as np
from core.beta import JacobianoCoherencia
app = Flask(__name__)
# Instanciamos el evaluador con los parámetros por defecto (threshold=0.85)
evaluador = JacobianoCoherencia()
@app.route('/api/v1/telemetry/evaluate', methods=['POST'])
def evaluate_matrix():
    data = request.get_json()
    if not data or 'matrix' not in data:
        return jsonify({'error': 'Payload inválido, se requiere el campo "matrix"'}), 400
    try:
        # Conversión de la lista entrante a una matriz de NumPy
        matriz_j = np.array(data['matrix'], dtype=float)
        # Validamos que sea una matriz cuadrada (ej. 3x3) para mantener consistencia con el core
        if matriz_j.shape != (3, 3):
            return jsonify({'error': 'La dimensión de la matriz debe ser estrictamente 3x3'}), 400
        # Ejecución de la lógica del Jacobiano
        alerta, factor = evaluador.evaluar_estabilidad(matriz_j)
        return jsonify({
            'status': 'processed',
            'is_stable': not alerta,
            'penalty_factor': factor,
            'matrix_norm_fro': float(np.linalg.norm(matriz_j, ord='fro'))
        }), 200
    except Exception as e:
        return jsonify({'error': f'Error interno al procesar los datos: {str(e)}'}), 500
if __name__ == '__main__':
    # Usamos el puerto de pruebas por defecto, habilitando el modo debug controlado
    app.run(host='127.0.0.1', port=5000, debug=True)
