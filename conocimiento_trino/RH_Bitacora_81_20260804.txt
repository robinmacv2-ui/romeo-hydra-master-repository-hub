import json
from pathlib import Path
from typing import Dict, Any, List
class ComplianceEngine:
    def __init__(self, rules_path: str = "rules.json"):
        self.rules_path = Path(rules_path)
        self.rules_data = self._load_rules()
        self.meta = self.rules_data["meta"]
        self.rules = self.rules_data["rules"]
        self.uma_value = self.meta["uma_value_mxn"]
        print(f"[SIMULACIÓN] Motor de cumplimiento cargado.")
        print(f"[SIMULACIÓN] Versión de reglas: {self.meta['rules_version']}")
        print(f"[SIMULACIÓN] Última revisión: {self.meta['last_reviewed']}")
        print(f"[SIMULACIÓN] Valor UMA actual: {self.uma_value} MXN")
        print("-" * 60)
    def _load_rules(self) -> Dict[str, Any]:
        """Carga el archivo de reglas. Falla explícitamente si hay problemas."""
        if not self.rules_path.exists():
            raise FileNotFoundError(
                f"[ERROR CRÍTICO] No se encontró el archivo de reglas: {self.rules_path}\n"
                "No se puede continuar sin rules.json."
            )
        try:
            with open(self.rules_path, "r", encoding="utf-8") as f:
                data = json.load(f)
        except json.JSONDecodeError as e:
            raise ValueError(
                f"[ERROR CRÍTICO] El archivo rules.json está mal formado: {e}"
            )
        except Exception as e:
            raise RuntimeError(
                f"[ERROR CRÍTICO] No se pudo leer rules.json: {e}"
            )
        # Validaciones básicas
        if "meta" not in data or "rules" not in data:
            raise ValueError("[ERROR CRÍTICO] rules.json no tiene la estructura esperada (meta + rules).")
        if "uma_value_mxn" not in data["meta"]:
            raise ValueError(
                "[ERROR CRÍTICO] Falta el valor 'uma_value_mxn' en meta. "
                "No se usará ningún valor por defecto silencioso."
            )
        return data
    def _resolve_threshold(self, condition: Dict) -> float:
        """Resuelve el valor del umbral (soporta referencias a UMA)."""
        value = condition["value"]
        if isinstance(value, str) and value == "uma_value_mxn":
            base = self.uma_value
            multiplier = condition.get("multiplier", 1)
            return base * multiplier
        return float(value)
    def _evaluate_condition(self, condition: Dict, transaction: Dict) -> bool:
        """Evalúa si una condición se cumple."""
        cond_type = condition["type"]
        if cond_type == "threshold":
            field = condition["field"]
            operator = condition["operator"]
            threshold = self._resolve_threshold(condition)
            if field not in transaction:
                return False  # Si falta el campo, la condición no se activa
            amount = float(transaction[field])
            if operator == "gte":
                return amount >= threshold
            elif operator == "gt":
                return amount > threshold
            elif operator == "lte":
                return amount <= threshold
            elif operator == "lt":
                return amount < threshold
            else:
                raise ValueError(f"Operador no soportado: {operator}")
        elif cond_type == "keyword_match":
            field = condition["field"]
            keywords = condition["keywords"]
            case_sensitive = condition.get("case_sensitive", False)
            if field not in transaction:
                return False
            text = str(transaction[field])
            if not case_sensitive:
                text = text.lower()
                keywords = [k.lower() for k in keywords]
            return any(keyword in text for keyword in keywords)
        else:
            raise ValueError(f"Tipo de condición no soportado: {cond_type}")
    def evaluate(self, transaction: Dict[str, Any]) -> Dict[str, Any]:
        """
        Evalúa una transacción contra todas las reglas.
        Retorna un resultado con estado y mensajes.
        """
        results = {
            "status": "APPROVED",
            "violations": [],
            "requires_review": False,
            "emulated_authority": "EMULATED_AUTHORITY",
            "simulation_note": "[SIMULACIÓN] Esta es una evaluación simulada. No tiene validez legal ni regulatoria real."
        }
        for rule in self.rules:
            condition_met = self._evaluate_condition(rule["condition"], transaction)
            if not condition_met:
                continue  # La regla no se activa
            action = rule["action"]
            message = rule["message_on_fail"]
            violation = {
                "rule_id": rule["id"],
                "rule_name": rule["name"],
                "severity": rule["severity"],
                "action": action,
                "message": message
            }
            if action == "reject":
                results["status"] = "REJECTED"
                results["violations"].append(violation)
            elif action == "requires_external_flag":
                flag_name = rule["external_flag"]
                flag_present = transaction.get(flag_name, False)
                if not flag_present:
                    results["status"] = "REJECTED"
                    results["violations"].append(violation)
            elif action == "requires_review":
                results["requires_review"] = True
                results["violations"].append(violation)
                # No cambia el status a REJECTED, solo marca para revisión
        return results
