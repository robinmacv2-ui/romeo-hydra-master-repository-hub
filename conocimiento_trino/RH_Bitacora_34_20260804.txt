import hashlib
import json
import time

class TarjetaLogicaQuantik:
    def __init__(self):
        # Matriz de Deducción Binaria Fraccionaria (+/- y -/+) según CÓDICE PPRH
        self.vectores_cardinales = {
            'S': {'+': 'Materia Luminosa (Activo)', '-': 'Materia Oscura (Latente)'},
            'O': {'+': 'Materia Luminosa (Activo)', '-': 'Materia Oscura (Latente)'},
            'N': {'-': 'Materia Luminosa (Activo)', '+': 'Materia Oscura (Latente)'},
            'I': {'-': 'Materia Luminosa (Activo)', '+': 'Materia Oscura (Latente)'}
        }
        self.nucleo_aislamiento_contenido = False

    def procesar_adn_digital(self, palabra_3bits):
        """
        Analiza la polaridad del paquete convexo-cuántico de entrada.
        Patrón 'Viborita' (predominancia de 0) -> Flujo Latente / Enrutamiento
        Patrón Activo (predominancia de 1)     -> Flujo Activo / Conmutación
        """
        conteo_unos = palabra_3bits.count('1')
        conteo_ceros = palabra_3bits.count('0')
        
        if conteo_ceros > conteo_unos:
            return "ADN Digital Negativo (Viborita)", "-"
        else:
            return "ADN Digital Positivo (Estructural)", "+"

    def colision_seis_haces(self, fase_polaridad):
        """
        Simula la convergencia de los 6 haces concurrentes en el centro geométrico.
        Bifurcación A: Refracción y salida periférica.
        Bifurcación B: Contención estática y autorreplicación (Final y se repite).
        """
        print("\n--- Impacto en el Núcleo de Aislamiento Central ---")
        print("Convergencia: Haz N, Haz I, Haz S, Haz O + 2 Haces Diagonales de la Matriz X.")
        
        if fase_polaridad == "+":
            print("[Bifurcación A Activada] -> Refracción de Materia Luminosa hacia canales periféricos.")
            self.nucleo_aislamiento_contenido = False
            return "Salida Conmutada Excitada"
        else:
            print("[Bifurcación B Activada] -> Interferencia Confinada (Materia Oscura).")
            print("[Efecto Espejo] -> La luz se contiene a sí misma. Persistencia geométrica activa.")
            self.nucleo_aislamiento_contenido = True
            return "Bucle Recursivo (Final y se repite)"

    def ejecutar_ciclo(self, secuencia_estados):
        print(f"=== INICIANDO NANOCHIP DE LÁSER LÓGICO MODULAR 1.0 ===")
        for estado in secuencia_estados:
            tipo_adn, polaridad = self.procesar_adn_digital(estado)
            print(f"\nEstado Procesado: [{estado}] -> Tipo: {tipo_adn} | Polaridad Asignada: {polaridad}")
            
            # Evaluación en los vectores según el CÓDICE PPRH
            for vector, fases in self.vectores_cardinales.items():
                regimen = fases[polaridad]
                # Modulación visual básica de la señal en consola
                simbolo_fase = "▲" if polaridad == "+" else "▼"
                print(f"  Vector {vector} [{simbolo_fase}]: Operando en régimen de {regimen}")
            
            # Simulación de colisión central
            resultado_nucleo = self.colision_seis_haces(polaridad)
            print(f"Resultado del Núcleo: {resultado_nucleo}")
            time.sleep(0.1)
