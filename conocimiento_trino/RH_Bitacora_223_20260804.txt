Generador de identidad criptográfica para E.E.R.H
Ejemplo seguro en Python usando Ed25519:
Archivo:
generate_master_key.py
Código:
from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey
from cryptography.hazmat.primitives import serialization
import os
import json
import hashlib
from datetime import datetime
PROJECT = "E.E.R.H-ROMEO-HYDRA-v3.0-RC1"
def generate_keypair():
    private_key = Ed25519PrivateKey.generate()
    public_key = private_key.public_key()
    private_bytes = private_key.private_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PrivateFormat.Raw,
        encryption_algorithm=serialization.NoEncryption()
    )
    public_bytes = public_key.public_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PublicFormat.Raw
    )
    return private_bytes, public_bytes
def create_identity():
    private_key, public_key = generate_keypair()
    fingerprint = hashlib.sha256(
        public_key
    ).hexdigest()
    identity = {
        "system": PROJECT,
        "created": datetime.utcnow().isoformat(),
        "algorithm": "Ed25519",
        "public_key": public_key.hex(),
        "fingerprint": fingerprint,
        "purpose": "ROOT_OF_TRUST"
    }
    os.makedirs("vault", exist_ok=True)
    with open(
        "vault/root_identity.json",
        "w"
    ) as f:
        json.dump(identity, f, indent=4)
    with open(
        "vault/root_private.key",
        "wb"
    ) as f:
        f.write(private_key)
    print("Identidad criptográfica creada")
    print("Fingerprint:")
    print(fingerprint)
if __name__ == "__main__":
    create_identity()

```python
import os
import json
import hashlib
from datetime import datetime
from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey
from cryptography.hazmat.primitives import serialization
class ERCA_Manager:
    """
    Gestor de la Autoridad Criptográfica Raíz (ERCA).
    Responsable de la identidad soberana del framework.
    """
    VAULT_PATH = "vault"
    IDENTITY_FILE = f"{VAULT_PATH}/root_identity.json"
    PRIVATE_KEY_FILE = f"{VAULT_PATH}/root_private.key"
    @classmethod
    def initialize_root(cls):
        """Genera el par de claves raíz si no existen."""
        if os.path.exists(cls.IDENTITY_FILE):
            print("[ERCA] Identidad ya existente. Operación abortada para proteger la raíz.")
            return False
        os.makedirs(cls.VAULT_PATH, exist_ok=True)
        private_key = Ed25519PrivateKey.generate()
        public_key = private_key.public_key()
        # Serialización de la clave privada
        priv_bytes = private_key.private_bytes(
            encoding=serialization.Encoding.Raw,
            format=serialization.PrivateFormat.Raw,
            encryption_algorithm=serialization.NoEncryption()
        )
        # Serialización de la clave pública
        pub_bytes = public_key.public_bytes(
            encoding=serialization.Encoding.Raw,
            format=serialization.PublicFormat.Raw
        )
        # Creación de identidad
        fingerprint = hashlib.sha256(pub_bytes).hexdigest()
        identity = {
            "system": "E.E.R.H-ROMEO-HYDRA-v3.0-RC1",
            "created": datetime.utcnow().isoformat(),
            "algorithm": "Ed25519",
            "public_key_hex": pub_bytes.hex(),
            "fingerprint": fingerprint,
            "status": "ROOT_OF_TRUST"
        }
        with open(cls.IDENTITY_FILE, "w") as f:
            json.dump(identity, f, indent=4)
        with open(cls.PRIVATE_KEY_FILE, "wb") as f:
            f.write(priv_bytes)
        print(f"[ERCA] Identidad Criptográfica generada. Fingerprint: {fingerprint}")
        return True
    @classmethod
    def get_identity(cls):
        if not os.path.exists(cls.IDENTITY_FILE):
            raise FileNotFoundError("ERCA no inicializada. Ejecute initialize_root().")
        with open(cls.IDENTITY_FILE, "r") as f:
            return json.load(f)
if __name__ == "__main__":
    ERCA_Manager.initialize_root()
```

### 1. Implementación: core/security/signer.py
Este módulo garantiza que cada *payload* (el estado del sistema) esté vinculado matemáticamente a una identidad privada.
```python
import json
from cryptography.hazmat.primitives.asymmetric import ed25519
from cryptography.hazmat.primitives import serialization
class Signer:
    def __init__(self, private_key_path: str):
        # Carga la clave privada desde tu entorno local (debe estar en /vault/)
        with open(private_key_path, "rb") as key_file:
            self._private_key = serialization.load_pem_private_key(
                key_file.read(),
                password=None
            )
    def canonicalize(self, payload: dict) -> bytes:
        """
        Ordena las claves del diccionario para garantizar consistencia 
        en la firma (hash determinístico).
        """
        return json.dumps(payload, sort_keys=True).encode("utf-8")
    def sign_payload(self, payload: dict) -> str:
        """
        Genera la firma digital para el payload canonicalizado.
        """
        data = self.canonicalize(payload)
        signature = self._private_key.sign(data)
        return signature.hex()
```
