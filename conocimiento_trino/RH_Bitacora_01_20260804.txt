import numpy as np
import time
class TraductorCapa1:
    def __init__(self, lambda_th=0.08):
        # Base ortonormal en R4
        self.eS = np.array([1/np.sqrt(2), 0, 0,  1/np.sqrt(2)], dtype=np.float32)
        self.eO = np.array([0, 1/np.sqrt(2), 0, -1/np.sqrt(2)], dtype=np.float32)
        self.eN = np.array([0, 0, 1, 0], dtype=np.float32)
        self.eI = np.array([1/np.sqrt(2), 0, 0, -1/np.sqrt(2)], dtype=np.float32)
        # Estados canónicos ortonormales
        self.v1 = (self.eS + self.eO + self.eI) / np.sqrt(3)
        self.v0 = (self.eO - self.eI) / np.sqrt(2)
        self.lambda_th = lambda_th
        self.P_W = np.array([True, True, False, True]) # Máscara para subespacio W
    def plegar_bloque(self, bits):
        N = bits.shape[0]
        out = np.empty((N, 4), dtype=np.float32)
        mask = bits.astype(bool)
        out[mask] = self.v1
        out[~mask] = self.v0
        return out
    def aislar(self, X):
        # 1. Proyectar sobre W (anular N temporalmente)
        Xw = X.copy()
        residuo_previo = Xw[:, 2]
        Xw[:, 2] = 0.0
        # 2. Soft-thresholding en componentes de W
        absXw = np.abs(Xw)
        Xw_st = np.sign(Xw) * np.maximum(absXw - self.lambda_th, 0.0)
        # 3. El drenaje: La pérdida de norma de W se suma a N
        # Calculamos cuánta energía se perdió en el thresholding
        diff = Xw - Xw_st
        drenaje = np.sum(diff**2, axis=1)
        Xw_st[:, 2] = residuo_previo + drenaje
        return Xw_st
class KernelSigma_Capa2:
    def __init__(self, c1):
        self.c1 = c1
        # w_perp: Complemento ortogonal en W (v1 x v0 simplificado)
        # Para este espacio R4, lo definimos para completar la base de W
        self.w_perp = np.array([0.408, -0.816, 0, 0.408], dtype=np.float32) 
    def _tau(self, X): return X @ self.c1.v1
    def NOT(self, X):
        c1 = X @ self.c1.v1
        c0 = X @ self.c1.v0
        cp = X @ self.w_perp
        Y = (c0[:, None] * self.c1.v1 + 
             c1[:, None] * self.c1.v0 + 
             cp[:, None] * self.w_perp)
        Y[:, 2] = X[:, 2] # Preserva sumidero
        return Y
    def AND(self, A, B):
        tau = self._tau(A) * self._tau(B)
        Y = tau[:, None] * self.c1.v1 + (1 - tau[:, None]) * self.c1.v0
        return self.c1.aislar(Y)

#!/usr/bin/env python3
"""
PPRH Codex – Núcleo Geométrico Coherente (versión de referencia)
================================================================
Principios axiomáticos aplicados:
1. Espacio base R^4 con base ortonormal {v1, v0, w_perp, eN}.
2. Toda operación lógica es una transformación que actúa sobre
   el subespacio activo W = span{v1, v0, w_perp} y deja eN
   como sumidero de entropía real (ruido), nunca de atenuación lógica.
3. Norma libre en W: la magnitud activa es exactamente |τ|.
4. Acumulación de N gobernada por suma hiperbólica (LAL = 1.0).
5. Colapso por comparación de proyecciones + índice de confianza
   que separa claridad lógica de pureza entrópica.
6. Métrica objetiva: Bit Error Rate (BER) entre bits originales
   y bits recuperados tras la cadena de colisiones.
Autoría de la formalización: Grok (xAI) – auditoría forense continua.
"""
import numpy as np
import time
import hashlib
import json
from typing import Dict, List, Tuple
np.random.seed(42)
# ------------------------------------------------------------
# 1. Construcción de la base ortonormal
# ------------------------------------------------------------
def build_orthonormal_frame() -> Dict[str, np.ndarray]:
    """
    Construye {v1, v0, w_perp, eN} garantizando:
        <vi, vj> = δij
    """
    # Vectores semilla en el hiperplano N=0
    a = np.array([1.0, 1.0, 0.0, 1.0], dtype=np.float64)
    b = np.array([0.0, 1.0, 0.0, -1.0], dtype=np.float64)
    v1 = a / np.linalg.norm(a)
    v0 = b - np.dot(b, v1) * v1
    v0 = v0 / np.linalg.norm(v0)
    # Complemento ortogonal dentro de W (producto cruz en las coords 0,1,3)
    cross = np.cross(v1[[0, 1, 3]], v0[[0, 1, 3]])
    w_perp = np.array([cross[0], cross[1], 0.0, cross[2]])
    w_perp /= np.linalg.norm(w_perp)
    eN = np.array([0.0, 0.0, 1.0, 0.0])
    # Verificaciones estrictas
    assert abs(np.dot(v1, v0)) < 1e-12
    assert abs(np.dot(v1, w_perp)) < 1e-12
    assert abs(np.dot(v0, w_perp)) < 1e-12
    assert abs(np.linalg.norm(v1) - 1.0) < 1e-12
    assert abs(np.linalg.norm(v0) - 1.0) < 1e-12
    assert abs(np.linalg.norm(w_perp) - 1.0) < 1e-12
    return {
        "v1": v1.astype(np.float32),
        "v0": v0.astype(np.float32),
        "w_perp": w_perp.astype(np.float32),
        "eN": eN.astype(np.float32),
    }
# ------------------------------------------------------------
# 2. Capa 1 – Traductor (Pliegue + Aislamiento)
# ------------------------------------------------------------
class Traductor:
    def __init__(self, frame: Dict[str, np.ndarray], lambda_th: float = 0.08):
        self.v1 = frame["v1"]
        self.v0 = frame["v0"]
        self.w_perp = frame["w_perp"]
        self.eN = frame["eN"]
        self.lambda_th = float(lambda_th)
        # Proyector ortogonal sobre W (anula la coordenada N)
        self.P_W = np.eye(4, dtype=np.float32)
        self.P_W[2, 2] = 0.0
    def fold(self, bits: np.ndarray) -> np.ndarray:
        """f : {0,1}^N → (R^4)^N   (vectorizado, O(N))"""
        out = np.empty((bits.shape[0], 4), dtype=np.float32)
        mask = bits.astype(bool)
        out[mask] = self.v1
        out[~mask] = self.v0
        return out
    def isolate(self, X: np.ndarray) -> np.ndarray:
        """
        Soft-threshold + drenaje del residual a eN.
        Solo se aplica en la frontera de entrada (Capa 1).
        """
        Xw = X @ self.P_W.T
        Xw_st = np.sign(Xw) * np.maximum(np.abs(Xw) - self.lambda_th, 0.0)
        residual = X - Xw_st
        Xw_st[:, 2] = residual.sum(axis=1)
        return Xw_st
# ------------------------------------------------------------
# 3. Capa 2 – Kernel Sigma (Álgebra de Colisión Geométrica)
# ------------------------------------------------------------
class KernelSigma:
    def __init__(self, frame: Dict[str, np.ndarray], LAL: float = 1.0):
        self.v1 = frame["v1"]
        self.v0 = frame["v0"]
        self.w_perp = frame["w_perp"]
        self.LAL = float(LAL)
        # Matrices de cambio de base (para claridad algebraica)
        # B = [v1 | v0 | w_perp | eN]
        self.B = np.column_stack([self.v1, self.v0, self.w_perp, frame["eN"]]).astype(np.float32)
        self.B_inv = self.B.T  # ortonormal ⇒ B^{-1} = B^T
    def _coords(self, X: np.ndarray) -> np.ndarray:
        """Proyecta el bloque a coordenadas de la base ortonormal."""
        return X @ self.B_inv.T   # (N,4) en orden [c1, c0, cp, cN]
    def _from_coords(self, C: np.ndarray) -> np.ndarray:
        """Reconstruye desde coordenadas de la base."""
        return C @ self.B.T
    def _hyperbolic_sum(self, a: np.ndarray, b: np.ndarray) -> np.ndarray:
        """Suma hiperbólica acotada por LAL (solo para el canal N)."""
        a = np.abs(a.astype(np.float64))
        b = np.abs(b.astype(np.float64))
        return ((a + b) / (1.0 + (a * b) / (self.LAL ** 2))).astype(np.float32)
    def NOT(self, X: np.ndarray) -> np.ndarray:
        """
        Isometría que intercambia v1 ↔ v0.
        Matriz en la base ortonormal:
            diag-block [[0,1],[1,0]] sobre (c1,c0), identidad en (cp,cN)
        """
        C = self._coords(X)
        C_new = C.copy()
        C_new[:, 0] = C[:, 1]   # c1 ← c0
        C_new[:, 1] = C[:, 0]   # c0 ← c1
        # cp y cN se preservan
        return self._from_coords(C_new)
    def AND(self, A: np.ndarray, B: np.ndarray) -> np.ndarray:
        """
        Operador bilineal de norma libre:
            τ = <A,v1> * <B,v1>
            Y_active = τ * v1
            N = hyperbolic_sum(N_A, N_B)
        """
        tau = (A @ self.v1) * (B @ self.v1)
        Y = tau[:, None] * self.v1
        Y[:, 2] = self._hyperbolic_sum(A[:, 2], B[:, 2])
        return Y
    def OR(self, A: np.ndarray, B: np.ndarray) -> np.ndarray:
        """
        Disyunción de norma libre:
            τ = τA + τB - τA*τB
            Y_active = τ * v1
            N = hyperbolic_sum(N_A, N_B)
        """
        tA = A @ self.v1
        tB = B @ self.v1
        tau = tA + tB - tA * tB
        Y = tau[:, None] * self.v1
        Y[:, 2] = self._hyperbolic_sum(A[:, 2], B[:, 2])
        return Y
    def health(self, X: np.ndarray) -> Dict[str, float]:
        C = self._coords(X)
        active_norm = np.linalg.norm(C[:, :3], axis=1)
        return {
            "active_norm_mean": float(np.mean(active_norm)),
            "active_norm_std": float(np.std(active_norm)),
            "N_mean": float(np.mean(np.abs(C[:, 3]))),
            "N_max": float(np.max(np.abs(C[:, 3]))),
            "orthogonality": float(abs(self.v1 @ self.v0)),
        }
# ------------------------------------------------------------
# 4. Capa 3 – Árbitro de Coherencia
# ------------------------------------------------------------
class Arbitro:
    def __init__(self, frame: Dict[str, np.ndarray], tau_crit: float = 0.45):
        self.v1 = frame["v1"]
        self.v0 = frame["v0"]
        self.tau_crit = float(tau_crit)
    def collapse(self, X: np.ndarray) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
        """
        Colapso geométrico:
            bit = 1  ⇔  <x,v1> > <x,v0>
        Confianza = claridad_lógica × pureza_entrópica
        """
        p1 = X @ self.v1
        p0 = X @ self.v0
        margin = p1 - p0
        bits = (margin > 0.0).astype(np.int32)
        # Claridad: qué tan dominante es la proyección activa
        clarity = np.abs(p1) / (np.abs(p1) + np.abs(p0) + 1e-8)
        # Pureza: distancia al techo de entropía
        purity = 1.0 - np.clip(np.abs(X[:, 2]) / self.tau_crit, 0.0, 1.0)
        confidence = clarity * purity
        return bits, confidence, margin
    def ledger_entries(self, X: np.ndarray, max_entries: int = 5) -> List[Dict]:
        bits, conf, margin = self.collapse(X)
        entries = []
        prev = "0" * 16
        for i in range(min(max_entries, X.shape[0])):
            state = "ESTABLE" if abs(X[i, 2]) < self.tau_crit else "RIESGO_ENTROPICO"
            h = hashlib.sha256(X[i].tobytes() + prev.encode()).hexdigest()[:16]
            e = {
                "id": i,
                "bit": int(bits[i]),
                "margin": round(float(margin[i]), 5),
                "confidence": round(float(conf[i]), 4),
                "N": round(float(abs(X[i, 2])), 4),
                "state": state,
                "hash": h,
            }
            if state != "ESTABLE":
                e["alert"] = "INCERTIDUMBRE_LOGICA"
            entries.append(e)
            prev = h
        return entries

#!/usr/bin/env python3
"""
PPRH Codex – Núcleo Geométrico Coherente (versión de referencia)
================================================================
Principios axiomáticos aplicados:
1. Espacio base R^4 con base ortonormal {v1, v0, w_perp, eN}.
2. Toda operación lógica es una transformación que actúa sobre
   el subespacio activo W = span{v1, v0, w_perp} y deja eN
   como sumidero de entropía real (ruido), nunca de atenuación lógica.
3. Norma libre en W: la magnitud activa es exactamente |τ|.
4. Acumulación de N gobernada por suma hiperbólica (LAL = 1.0).
5. Colapso por comparación de proyecciones + índice de confianza
   que separa claridad lógica de pureza entrópica.
6. Métrica objetiva: Bit Error Rate (BER) entre bits originales
   y bits recuperados tras la cadena de colisiones.
Autoría de la formalización: Grok (xAI) – auditoría forense continua.
"""
import numpy as np
import time
import hashlib
import json
from typing import Dict, List, Tuple
np.random.seed(42)
# ------------------------------------------------------------
# 1. Construcción de la base ortonormal
# ------------------------------------------------------------
def build_orthonormal_frame() -> Dict[str, np.ndarray]:
    """
    Construye {v1, v0, w_perp, eN} garantizando:
        <vi, vj> = δij
    """
    # Vectores semilla en el hiperplano N=0
    a = np.array([1.0, 1.0, 0.0, 1.0], dtype=np.float64)
    b = np.array([0.0, 1.0, 0.0, -1.0], dtype=np.float64)
    v1 = a / np.linalg.norm(a)
    v0 = b - np.dot(b, v1) * v1
    v0 = v0 / np.linalg.norm(v0)
    # Complemento ortogonal dentro de W (producto cruz en las coords 0,1,3)
    cross = np.cross(v1[[0, 1, 3]], v0[[0, 1, 3]])
    w_perp = np.array([cross[0], cross[1], 0.0, cross[2]])
    w_perp /= np.linalg.norm(w_perp)
    eN = np.array([0.0, 0.0, 1.0, 0.0])
    # Verificaciones estrictas
    assert abs(np.dot(v1, v0)) < 1e-12
    assert abs(np.dot(v1, w_perp)) < 1e-12
    assert abs(np.dot(v0, w_perp)) < 1e-12
    assert abs(np.linalg.norm(v1) - 1.0) < 1e-12
    assert abs(np.linalg.norm(v0) - 1.0) < 1e-12
    assert abs(np.linalg.norm(w_perp) - 1.0) < 1e-12
    return {
        "v1": v1.astype(np.float32),
        "v0": v0.astype(np.float32),
        "w_perp": w_perp.astype(np.float32),
        "eN": eN.astype(np.float32),
    }
# ------------------------------------------------------------
# 2. Capa 1 – Traductor (Pliegue + Aislamiento)
# ------------------------------------------------------------
class Traductor:
    def __init__(self, frame: Dict[str, np.ndarray], lambda_th: float = 0.08):
        self.v1 = frame["v1"]
        self.v0 = frame["v0"]
        self.w_perp = frame["w_perp"]
        self.eN = frame["eN"]
        self.lambda_th = float(lambda_th)
        # Proyector ortogonal sobre W (anula la coordenada N)
        self.P_W = np.eye(4, dtype=np.float32)
        self.P_W[2, 2] = 0.0
    def fold(self, bits: np.ndarray) -> np.ndarray:
        """f : {0,1}^N → (R^4)^N   (vectorizado, O(N))"""
        out = np.empty((bits.shape[0], 4), dtype=np.float32)
        mask = bits.astype(bool)
        out[mask] = self.v1
        out[~mask] = self.v0
        return out
    def isolate(self, X: np.ndarray) -> np.ndarray:
        """
        Soft-threshold + drenaje del residual a eN.
        Solo se aplica en la frontera de entrada (Capa 1).
        """
        Xw = X @ self.P_W.T
        Xw_st = np.sign(Xw) * np.maximum(np.abs(Xw) - self.lambda_th, 0.0)
        residual = X - Xw_st
        Xw_st[:, 2] = residual.sum(axis=1)
        return Xw_st
# ------------------------------------------------------------
# 3. Capa 2 – Kernel Sigma (Álgebra de Colisión Geométrica)
# ------------------------------------------------------------
class KernelSigma:
    def __init__(self, frame: Dict[str, np.ndarray], LAL: float = 1.0):
        self.v1 = frame["v1"]
        self.v0 = frame["v0"]
        self.w_perp = frame["w_perp"]
        self.LAL = float(LAL)
        # Matrices de cambio de base (para claridad algebraica)
        # B = [v1 | v0 | w_perp | eN]
        self.B = np.column_stack([self.v1, self.v0, self.w_perp, frame["eN"]]).astype(np.float32)
        self.B_inv = self.B.T  # ortonormal ⇒ B^{-1} = B^T
    def _coords(self, X: np.ndarray) -> np.ndarray:
        """Proyecta el bloque a coordenadas de la base ortonormal."""
        return X @ self.B_inv.T   # (N,4) en orden [c1, c0, cp, cN]
    def _from_coords(self, C: np.ndarray) -> np.ndarray:
        """Reconstruye desde coordenadas de la base."""
        return C @ self.B.T
    def _hyperbolic_sum(self, a: np.ndarray, b: np.ndarray) -> np.ndarray:
        """Suma hiperbólica acotada por LAL (solo para el canal N)."""
        a = np.abs(a.astype(np.float64))
        b = np.abs(b.astype(np.float64))
        return ((a + b) / (1.0 + (a * b) / (self.LAL ** 2))).astype(np.float32)
    def NOT(self, X: np.ndarray) -> np.ndarray:
        """
        Isometría que intercambia v1 ↔ v0.
        Matriz en la base ortonormal:
            diag-block [[0,1],[1,0]] sobre (c1,c0), identidad en (cp,cN)
        """
        C = self._coords(X)
        C_new = C.copy()
        C_new[:, 0] = C[:, 1]   # c1 ← c0
        C_new[:, 1] = C[:, 0]   # c0 ← c1
        # cp y cN se preservan
        return self._from_coords(C_new)
    def AND(self, A: np.ndarray, B: np.ndarray) -> np.ndarray:
        """
        Operador bilineal de norma libre:
            τ = <A,v1> * <B,v1>
            Y_active = τ * v1
            N = hyperbolic_sum(N_A, N_B)
        """
        tau = (A @ self.v1) * (B @ self.v1)
        Y = tau[:, None] * self.v1
        Y[:, 2] = self._hyperbolic_sum(A[:, 2], B[:, 2])
        return Y
    def OR(self, A: np.ndarray, B: np.ndarray) -> np.ndarray:
        """
        Disyunción de norma libre:
            τ = τA + τB - τA*τB
            Y_active = τ * v1
            N = hyperbolic_sum(N_A, N_B)
        """
        tA = A @ self.v1
        tB = B @ self.v1
        tau = tA + tB - tA * tB
        Y = tau[:, None] * self.v1
        Y[:, 2] = self._hyperbolic_sum(A[:, 2], B[:, 2])
        return Y
    def health(self, X: np.ndarray) -> Dict[str, float]:
        C = self._coords(X)
        active_norm = np.linalg.norm(C[:, :3], axis=1)
        return {
            "active_norm_mean": float(np.mean(active_norm)),
            "active_norm_std": float(np.std(active_norm)),
            "N_mean": float(np.mean(np.abs(C[:, 3]))),
            "N_max": float(np.max(np.abs(C[:, 3]))),
            "orthogonality": float(abs(self.v1 @ self.v0)),
        }
# ------------------------------------------------------------
# 4. Capa 3 – Árbitro de Coherencia
# ------------------------------------------------------------
class Arbitro:
    def __init__(self, frame: Dict[str, np.ndarray], tau_crit: float = 0.45):
        self.v1 = frame["v1"]
        self.v0 = frame["v0"]
        self.tau_crit = float(tau_crit)
    def collapse(self, X: np.ndarray) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
        """
        Colapso geométrico:
            bit = 1  ⇔  <x,v1> > <x,v0>
        Confianza = claridad_lógica × pureza_entrópica
        """
        p1 = X @ self.v1
        p0 = X @ self.v0
        margin = p1 - p0
        bits = (margin > 0.0).astype(np.int32)
        # Claridad: qué tan dominante es la proyección activa
        clarity = np.abs(p1) / (np.abs(p1) + np.abs(p0) + 1e-8)
        # Pureza: distancia al techo de entropía
        purity = 1.0 - np.clip(np.abs(X[:, 2]) / self.tau_crit, 0.0, 1.0)
        confidence = clarity * purity
        return bits, confidence, margin
    def ledger_entries(self, X: np.ndarray, max_entries: int = 5) -> List[Dict]:
        bits, conf, margin = self.collapse(X)
        entries = []
        prev = "0" * 16
        for i in range(min(max_entries, X.shape[0])):
            state = "ESTABLE" if abs(X[i, 2]) < self.tau_crit else "RIESGO_ENTROPICO"
            h = hashlib.sha256(X[i].tobytes() + prev.encode()).hexdigest()[:16]
            e = {
                "id": i,
                "bit": int(bits[i]),
                "margin": round(float(margin[i]), 5),
                "confidence": round(float(conf[i]), 4),
                "N": round(float(abs(X[i, 2])), 4),
                "state": state,
                "hash": h,
            }
            if state != "ESTABLE":
                e["alert"] = "INCERTIDUMBRE_LOGICA"
            entries.append(e)
            prev = h
        return entries

def run_full_validation(N: int = 1_000_000, sigma: float = 0.1, lambda_th: float = 0.08):

    print("=" * 72)
    frame = build_orthonormal_frame()
    trad = Traductor(frame, lambda_th=lambda_th)
    kern = KernelSigma(frame, LAL=1.0)
    arb = Arbitro(frame, tau_crit=0.45)
    # --- Datos de entrada ---
    t0 = time.perf_counter()
    bits_orig = np.random.randint(0, 2, size=N, dtype=np.int32)
    noise = np.random.normal(0.0, sigma, size=(N, 4)).astype(np.float32)
    # --- Capa 1 ---
    t1 = time.perf_counter()
    T = trad.fold(bits_orig)
    T_noisy = T + noise
    T_clean = trad.isolate(T_noisy)
    t2 = time.perf_counter()
    h1 = kern.health(T_clean)
    drain = 100.0 * np.mean(np.minimum(
        np.abs(T_clean[:, 2]) / (np.linalg.norm(noise, axis=1) + 1e-12), 1.0))
    print("\n[CAPA 1 – Pliegue + Aislamiento]")

    print(f"  Drenaje de ruido → N       : {drain:.1f} %")
    print(f"  Latencia                   : {(t2-t1)*1e3:.1f} ms")
    # --- Capa 2 – cadena de colisiones ---
    t3 = time.perf_counter()
    A = T_clean
    B = np.roll(A, shift=N // 3, axis=0)          # segundo operando desfasado
    R = kern.NOT(A)
    R = kern.AND(R, B)
    R = kern.OR(R, A)
    R = kern.NOT(R)
    t4 = time.perf_counter()
    h2 = kern.health(R)
    mops = (4 * N) / max(t4 - t3, 1e-9) / 1e6

    print(f"  Throughput                 : {mops:.1f} Mops/s")
    print(f"  Latencia                   : {(t4-t3)*1e3:.1f} ms")
    # --- Capa 3 – Colapso + fidelidad ---
    t5 = time.perf_counter()
    bits_rec, conf, _ = arb.collapse(R)
    t6 = time.perf_counter()
    # Bit Error Rate objetivo
    # Como la cadena es NOT→AND→OR→NOT sobre datos desfasados,
    # la semántica esperada no es la identidad. Medimos
    # fidelidad del colapso sobre el estado final vs. una
    # referencia lógica simple (bit original) solo como
    # indicador de estabilidad del colapso, no de corrección
    # funcional de la cadena.
    ber = np.mean(bits_rec != bits_orig)
    conf_mean = float(np.mean(conf))
    conf_median = float(np.median(conf))
    risk_pct = 100.0 * np.mean(np.abs(R[:, 2]) >= arb.tau_crit)
    print("\n[CAPA 3 – Árbitro de Coherencia]")
    print(f"  Confianza media / mediana  : {conf_mean:.4f} / {conf_median:.4f}")
    print(f"  % RIESGO_ENTROPICO         : {risk_pct:.2f} %")
    print(f"  BER (colapso vs original)  : {ber*100:.2f} %")
    print(f"  Latencia                   : {(t6-t5)*1e3:.1f} ms")
    # Muestra del Delta Ledger
    print("\n[DELTA LEDGER – muestra]")
    print(json.dumps(arb.ledger_entries(R, max_entries=3), indent=2, ensure_ascii=False))
    total = t6 - t0
    print("\n" + "=" * 72)
    print("RESUMEN DE COHERENCIA")
    print(f"  Tiempo total wall-clock    : {total*1e3:.1f} ms")
    print(f"  Throughput E2E             : {N/total/1e6:.2f} Mbits/s")
    print(f"  N acotado por LAL          : {'SÍ' if h2['N_max'] <= 1.0001 else 'NO'} (máx={h2['N_max']:.4f})")
    print(f"  Ortogonalidad preservada   : {'SÍ' if h2['orthogonality'] < 1e-6 else 'NO'}")

        "mops": mops,
        "confidence_mean": conf_mean,
