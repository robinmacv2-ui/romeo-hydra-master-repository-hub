# CAPA 1: TRADUCTOR UNIVERSAL (GEOMETRÍA)
# ==========================================
class TraductorCapa1:
    def __init__(self, lambda_th=0.08):
        # Base ortonormal estricta (Gram-Schmidt verificado)
        # Espacio R4: [Dim1, Dim2, N (Sumidero), Dim3]
        self.v1     = np.array([ 1.0,  0.0, 0.0, 0.0], dtype=np.float32) # Fase Activa (Materia Luminosa)
        self.v0     = np.array([ 0.0,  1.0, 0.0, 0.0], dtype=np.float32) # Fase Latente (Materia Oscura)
        self.eN     = np.array([ 0.0,  0.0, 1.0, 0.0], dtype=np.float32) # Sumidero Termodinámico
        self.w_perp = np.array([ 0.0,  0.0, 0.0, 1.0], dtype=np.float32) # Complemento Ortogonal
        
        # Validar ortogonalidad de la matriz de transposición
        assert np.abs(np.dot(self.v1, self.v0)) < 1e-6, "Fallo crítico: Base no ortogonal."
        
        self.lambda_th = lambda_th

    def plegar_bloque(self, bits: np.ndarray) -> np.ndarray:
        """Bifurcación 1->4 Vectorizada"""
        N = bits.shape[0]
        out = np.zeros((N, 4), dtype=np.float32)
        mask = bits.astype(bool)
        out[mask] = self.v1
        out[~mask] = self.v0
        return out

    def aislar(self, X: np.ndarray) -> np.ndarray:
        """Aislamiento estadístico: Drenaje de entropía hacia N"""
        # Extraer coordenadas en la base ortonormal
        c1 = X @ self.v1
        c0 = X @ self.v0
        cp = X @ self.w_perp
        cn_previo = X @ self.eN
        
        # Soft-thresholding sobre componentes activos
        c1_st = np.sign(c1) * np.maximum(np.abs(c1) - self.lambda_th, 0.0)
        c0_st = np.sign(c0) * np.maximum(np.abs(c0) - self.lambda_th, 0.0)
        cp_st = np.sign(cp) * np.maximum(np.abs(cp) - self.lambda_th, 0.0)
        
        # Drenaje exacto de la energía filtrada hacia N
        drenaje = (c1 - c1_st)**2 + (c0 - c0_st)**2 + (cp - cp_st)**2
        cn_nuevo = cn_previo + drenaje
        
        # Reconstrucción geométrica
        Y = (c1_st[:, None] * self.v1 + 
             c0_st[:, None] * self.v0 + 
             cp_st[:, None] * self.w_perp + 
             cn_nuevo[:, None] * self.eN)
        return Y

# CAPA 2: KERNEL SIGMA (OPERADORES DE COLISIÓN)
# ==========================================
class KernelSigma_Capa2:
    def __init__(self, traductor):
        self.c1 = traductor
        self.LAL = 1.0 # Límite de Luis Ángel (Asíntota termodinámica)

    def _suma_hiperbolica(self, N_A: np.ndarray, N_B: np.ndarray) -> np.ndarray:
        """Previene la divergencia lineal saturando en LAL=1.0"""
        return (N_A + N_B) / (1.0 + (N_A * N_B) / (self.LAL**2))

    def NOT(self, X: np.ndarray) -> np.ndarray:
        """Isometría perfecta: Permutación ortogonal sin alterar el sumidero"""
        c1 = X @ self.c1.v1
        c0 = X @ self.c1.v0
        cp = X @ self.c1.w_perp
        cn = X @ self.c1.eN
        
        # Intercambio c1 <-> c0
        Y = (c0[:, None] * self.c1.v1 + 
             c1[:, None] * self.c1.v0 + 
             cp[:, None] * self.c1.w_perp + 
             cn[:, None] * self.c1.eN)
        return Y

    def AND(self, A: np.ndarray, B: np.ndarray) -> np.ndarray:
        """Colisión bilineal de norma libre"""
        tau_A = A @ self.c1.v1
        tau_B = B @ self.c1.v1
        tau_out = tau_A * tau_B
        
        Y = tau_out[:, None] * self.c1.v1
        Y += self._suma_hiperbolica(A @ self.c1.eN, B @ self.c1.eN)[:, None] * self.c1.eN
        return self.c1.aislar(Y)

# CAPA 3: ÁRBITRO DE COHERENCIA (COLAPSO)
# ==========================================
class ArbitroCoherencia_Capa3:
    def __init__(self, tau_critico=0.5):
        self.tau_critico = tau_critico

    def colapsar_y_auditar(self, tensores, traductor):
        # 1. Proyecciones
        c1 = tensores @ traductor.v1
        c0 = tensores @ traductor.v0
        
        # 2. Veredicto Geométrico Robusto
        bits_finales = (c1 > c0).astype(int)
        margen = np.abs(c1 - c0)
        
        # 3. SNR Topológico (Atenuación vs Entropía)
        norma_activa = np.maximum(np.abs(c1), np.abs(c0))
        entropia_n = np.abs(tensores @ traductor.eN)
        confianza = norma_activa / (norma_activa + entropia_n + 1e-9)
        
        # 4. Auditoría Delta Ledger
        riesgo_mask = (entropia_n > self.tau_critico) & (confianza < 0.5)
        
        muestra_ledger = []
        for i in range(min(3, tensores.shape[0])):
            estado = "RIESGO_ENTROPICO" if riesgo_mask[i] else "ESTABLE"
            hash_tensor = hashlib.sha256(tensores[i].tobytes()).hexdigest()
            
            bloque = {
                "arquitectura": DOI_SOFTWARE,
                "timestamp_id": i,
                "veredicto": int(bits_finales[i]),
                "confianza": round(float(confianza[i]), 4),
                "entropia_n": round(float(entropia_n[i]), 4),
                "estado_integridad": estado,
                "hash_sha256": hash_tensor
            }
            muestra_ledger.append(bloque)
            
        return bits_finales, entropia_n, confianza, riesgo_mask, muestra_ledger
