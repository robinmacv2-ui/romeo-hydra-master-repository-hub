def registrar_en_ledger(payload_vector, respuesta_original, estado_transaccion):
    """
    Simula el Regulatory Evidence Arbitration Kernel (RAEK-1.0-MX).
    Registra la inmutabilidad de la decisión del modelo en el Delta Ledger.
    """
    try:
        # Estructura del bloque conforme a principios de auditoría externa
        timestamp = datetime.datetime.now(datetime.timezone.utc).isoformat()
        bloque_datos = {
            "timestamp": timestamp,
            "vector_payload": payload_vector,
            "original_respuesta": respuesta_original,
            "is_estado": estado_transaccion
        }
        # Generación del hash criptográfico del bloque
        hash_objeto = hashlib.sha256(json.dumps(bloque_datos, sort_keys=True).encode())
        bloque_hash = hash_objeto.hexdigest()
        print("\n--- INICIANDO BANCO DE PRUEBAS ROMEO-HYDRA ---")
        print("[ESTADO] Transmitiendo carga útil hacia el nodo local...")
        print("[EXITO] Respuesta del nodo local procesada.")
        print(f"[SUCCESS] Delta Ledger actualizado. Hash de bloque: {bloque_hash}...")
        return bloque_hash
    except Exception as e:
        print(f"\n[EXCEPCION CRITICA] Error en el registro del Ledger: {e}")
        return None

import json
import hashlib
from datetime import datetime
from pathlib import Path
import logging
logger = logging.getLogger(__name__)
class DeltaLedger:
    def __init__(self, data_dir="data"):
        self.data_dir = Path(data_dir)
        self.data_dir.mkdir(exist_ok=True)
        self.ledger_file = self.data_dir / "delta_ledger.json"
        self.chain = self._load_chain()
    def _load_chain(self):
        if self.ledger_file.exists():
            try:
                with open(self.ledger_file, 'r', encoding='utf-8') as f:
                    return json.load(f)
            except Exception:
                return []
        return []
    def _save_chain(self):
        with open(self.ledger_file, 'w', encoding='utf-8') as f:
            json.dump(self.chain, f, indent=2, ensure_ascii=False)
    def add_block(self, payload: dict, model_response: str, status: str = "APROBADO"):
        previous_hash = self.chain[-1]["hash"] if self.chain else "0" * 64
        block = {
            "index": len(self.chain) + 1,
            "timestamp": datetime.utcnow().isoformat(),
            "payload": payload,
            "model_response": model_response,
            "status": status,
            "previous_hash": previous_hash,
        }
        block_string = json.dumps(block, sort_keys=True).encode('utf-8')
        block["hash"] = hashlib.sha256(block_string).hexdigest()
        self.chain.append(block)
        self._save_chain()
        print(f"\n[SUCCESS] Delta Ledger actualizado.")
        print(f"Hash de bloque: {block['hash']}")
        return block

class OllamaClient:
    def __init__(self, base_url="http://localhost:11434"):
        self.base_url = base_url
    def generate(self, prompt: str, model: str = "llama3.2"):
        try:
            payload = {
                "model": model,
                "prompt": prompt,
                "stream": False,
                "temperature": 0.7
            }
            response = requests.post(f"{self.base_url}/api/generate", json=payload, timeout=60)
            response.raise_for_status()
            return response.json().get("response", "Sin respuesta")
        except:
            return "[SIMULADO] Respuesta del modelo local (Ollama no detectado)."

import sqlite3
import hashlib
import json
from datetime import datetime
from pathlib import Path
import logging
logger = logging.getLogger(__name__)
class DeltaLedger:
    def __init__(self, db_path="data/delta_ledger.db"):
        self.db_path = Path(db_path)
        self.db_path.parent.mkdir(exist_ok=True)
        self.conn = sqlite3.connect(self.db_path)
        self._create_table()
    def _create_table(self):
        with self.conn:
            self.conn.execute('''
                CREATE TABLE IF NOT EXISTS blocks (
                    id INTEGER PRIMARY KEY,
                    timestamp TEXT,
                    payload TEXT,
                    model_response TEXT,
                    status TEXT,
                    hash TEXT UNIQUE,
                    previous_hash TEXT
                )
            ''')
    def add_block(self, payload: dict, model_response: str, status: str = "APROBADO"):
        previous_hash = self._get_last_hash()
        block = {
            "timestamp": datetime.utcnow().isoformat(),
            "payload": payload,
            "model_response": model_response,
            "status": status,
            "previous_hash": previous_hash
        }
        block_string = json.dumps(block, sort_keys=True).encode('utf-8')
        block_hash = hashlib.sha256(block_string).hexdigest()
        with self.conn:
            self.conn.execute('''
                INSERT INTO blocks 
                (timestamp, payload, model_response, status, hash, previous_hash)
                VALUES (?, ?, ?, ?, ?, ?)
            ''', (
                block["timestamp"],
                json.dumps(payload),
                model_response,
                status,
                block_hash,
                previous_hash
            ))
        print(f"\n[SUCCESS] Bloque guardado en SQLite")
        print(f"Hash: {block_hash}")
        return block_hash
    def _get_last_hash(self):
        cursor = self.conn.execute("SELECT hash FROM blocks ORDER BY id DESC LIMIT 1")
        result = cursor.fetchone()
        return result[0] if result else "0" * 64
    def get_all_blocks(self):
        cursor = self.conn.execute("SELECT * FROM blocks ORDER BY id")
        return cursor.fetchall()
